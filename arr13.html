#include <stdio.h>
void main()
{
int arr[5], i, min;
printf("Enter 5 elements:\n");
for(i = 0; i < 5; i++)
{
scanf("%d", &arr[i]);
}
min = arr[0];
for(i = 1; i < 5; i++)
{
if(arr[i] < min)
{
min = arr[i];
}
}
printf("Minimum value = %d", min);
}

Output

Enter 5 elements:
10 50 30 80 20
Minimum value = 10
