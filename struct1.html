#include <stdio.h>
#include <conio.h>
struct employee
{
int id;
char name[30];
char designation[30];
};
void main()
{
struct employee e[50];
int i;
clrscr();
// Input employee details
for(i = 0; i < 50; i++)
{
printf("\nEnter details of employee %d", i+1);
printf("\nEnter Employee ID: ");
scanf("%d", &e[i].id);
printf("Enter Employee Name: ");
scanf("%s", e[i].name);
printf("Enter Designation: ");
scanf("%s", e[i].designation);
}
// Display employee details
printf("\n\nEMPLOYEE DETAILS");
printf("\n-----------------------------------------------");
printf("\nID\tNAME\t\tDESIGNATION");
printf("\n-----------------------------------------------");
for(i = 0; i < 50; i++)
{
printf("\n%d\t%s\t\t%s",
e[i].id, e[i].name, e[i].designation);
}
getch();
}

#include <stdio.h>
#include <conio.h>
struct student
{
int roll;
char name[30];
float fee;
char dob[15];
};
void main()
{
struct student s[50], temp;
int i, j, n;
clrscr();
printf("Enter number of students: ");
scanf("%d",&n);
for(i=0; i {
printf("\nEnter details of student %d",i+1);
printf("\nEnter Roll Number: ");
scanf("%d",&s[i].roll);
printf("Enter Name: ");
scanf("%s",s[i].name);
printf("Enter Fee: ");
scanf("%f",&s[i].fee);
printf("Enter DOB (DD-MM-YYYY): ");
scanf("%s",s[i].dob);
}
/* Sorting in ascending order */
for(i=0; i {
for(j=0; j {
if(s[j].roll > s[j+1].roll)
{
temp = s[j];
s[j] = s[j+1];
s[j+1] = temp;
}
}
}
printf("\n\nSTUDENT DETAILS IN ASCENDING ORDER");
printf("\nROLL\tNAME\tFEE\tDOB");
for(i=0; i {
printf("\n%d\t%s\t%.2f\t%s",
s[i].roll, s[i].name, s[i].fee, s[i].dob);
}
getch();
}
