#include <stdio.h>

int main()
{
int a[100], b[100], n, i;
printf("Enter the size of array: ");
scanf("%d", &n);
printf("Enter %d elements:\n", n);
for(i = 0; i < n; i++)
{
scanf("%d", &a[i]);
}
for(i = 0; i < n; i++)
{
b[i] = a[n - 1 - i];
}
printf("Second array in reverse order:\n");
for(i = 0; i < n; i++)
{
printf("%d ", b[i]);
}
}

Output

Enter the size of array: 5
Enter 5 elements:
10 20 30 40 50
Second array in reverse order:
50 40 30 20 10
