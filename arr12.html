#include <stdio.h>
void main()
{
int arr[5], i, max;
printf("Enter 5 elements:\n");
for(i = 0; i < 5; i++)
{
scanf("%d", &arr[i]);
}
max = arr[0];
for(i = 1; i < 5; i++)
{
if(arr[i] > max)
{
max = arr[i];
}
}
printf("Maximum value = %d", max);
}

Output

Enter 5 elements:
10 50 30 80 20
Maximum value = 80
