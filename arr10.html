#include <stdio.h>
void main()
{
int arr[5], i, num, found = 0;
printf("Enter 5 elements:\n");
for(i = 0; i < 5; i++)
{
scanf("%d", &arr[i]);
}
printf("Enter value to search: ");
scanf("%d", &num);
for(i = 0; i < 5; i++)
{
if(arr[i] == num)
{
found = 1;
break;
}
}
if(found == 1)
{
printf("Found");
}
else
{
printf("Not Found");
}
}

Output

Enter 5 elements:
10 20 30 40 50
Enter value to search: 30
Found
