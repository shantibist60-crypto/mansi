#include <stdio.h>
void swap(int *a, int *b)
{
int temp;
temp = *a;
*a = *b;
*b = temp;
}
void main()
{
int x, y;
printf("Enter two numbers: ");
scanf("%d %d", &x, &y);
printf("\nBefore swapping:");
printf("\nX = %d", x);
printf("\nY = %d", y);
swap(&x, &y);
printf("\n\nAfter swapping:");
printf("\nX = %d", x);
printf("\nY = %d", y);
}

Output

Enter two numbers: 10 20
Before swapping:
X = 10
Y = 20
After swapping:
X = 20
Y = 10
