#include <stdio.h>
void swap(int a, int b)
{
int temp;
temp = a;
a = b;
b = temp;
printf("\nInside swap function:");
printf("\nA = %d", a);
printf("\nB = %d", b);
}
void main()
{
int x, y;
printf("Enter two numbers: ");
scanf("%d %d", &x, &y);
printf("\nBefore swapping:");
printf("\nX = %d", x);
printf("\nY = %d", y);
swap(x, y);
printf("\n\nAfter calling function:");
printf("\nX = %d", x);
printf("\nY = %d", y);
}

Output

Enter two numbers: 10 20
Before swapping:
X = 10
Y = 20
Inside swap function:
A = 20
B = 10
After calling function:
X = 10
Y = 20
