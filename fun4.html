1. Factorial using Local and Global Variables

#include <stdio.h>
int fact = 1; // Global variable
void main()
{
int n, i; // Local variables
printf("Enter a number: ");
scanf("%d", &n);
for(i = 1; i <= n; i++)
{
fact = fact * i;
}
printf("Factorial = %d", fact);
}

Output

Enter a number: 5
Factorial = 120




2. Reverse a Number using Local and Global Variables

#include <stdio.h>
int rev = 0; // Global variable
void main()
{
int n, rem; // Local variables
printf("Enter a number: ");
scanf("%d", &n);
while(n != 0)
{
rem = n % 10;
rev = rev * 10 + rem;
n = n / 10;
}
printf("Reverse = %d", rev);
}

Output

Enter a number: 1234
Reverse = 4321




3. Palindrome Number using Local and Global Variables

#include <stdio.h>
int rev = 0; // Global variable
void main()
{
int n, temp, rem; // Local variables
printf("Enter a number: ");
scanf("%d", &n);
temp = n;
while(n != 0)
{ rem = n % 10;
rev = rev * 10 + rem;
n = n / 10;
}
if(temp == rev)
printf("Palindrome Number");
else
printf("Not a Palindrome Number");
}

Output

Enter a number: 121
Palindrome Number




4. Prime Number using Local and Global Variables

#include <stdio.h>
int count = 0; // Global variable
void main()
{
int n, i; // Local variables
printf("Enter a number: ");
scanf("%d", &n);
for(i = 1; i <= n; i++)
{
if(n % i == 0)
{
count++;
}
}
if(count == 2)
printf("Prime Number");
else
printf("Not a Prime Number");
}

Output

Enter a number: 7
Prime Number




5. Armstrong Number using Local and Global Variables

#include <stdio.h>
int sum = 0; // Global variable
void main()
{
int n, temp, rem; // Local variables
printf("Enter a number: ");
scanf("%d", &n);
temp = n;
while(n != 0)
{
rem = n % 10;
sum = sum + (rem * rem * rem);
n = n / 10;
}
if(temp == sum)
printf("Armstrong Number");
else
printf("Not an Armstrong Number");
}

Output

Enter a number: 153
Armstrong Number




6. Fibonacci Series using Local and Global Variables

#include <stdio.h>
int a = 0, b = 1; // Global variables
void main()
{
int n, i, c; // Local variables
printf("Enter number of terms: ");
scanf("%d", &n);
printf("Fibonacci Series: ");
for(i = 1; i <= n; i++)
{
printf("%d ", a);
c = a + b;
a = b;
b = c;
}
}

Output

Enter number of terms: 7
Fibonacci Series: 0 1 1 2 3 5 8
