#include<stdio.h>
#include<conio.h>
void main()
{
int a[2][2], b[2][2], c[2][2];
int i, j;
clrscr();
printf("Enter first matrix:\n");
for(i=0; i<2; i++)
{
for(j=0; j<2; j++)
{
scanf("%d", &a[i][j]);
}
}
printf("Enter second matrix:\n");
for(i=0; i<2; i++)
{
for(j=0; j<2; j++)
{
scanf("%d", &b[i][j]);
}
}
// Subtraction of matrices
for(i=0; i<2; i++)
{
for(j=0; j<2; j++)
{
c[i][j] = a[i][j] - b[i][j];
}
}
printf("\nSubtraction of two matrices:\n");
for(i=0; i<2; i++)
{
for(j=0; j<2; j++)
{
printf("%d\t", c[i][j]);
}
printf("\n");
}
getch();
}
